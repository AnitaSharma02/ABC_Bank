INSTALLATION -
----------------------------------------------------------------------------
1. First remove these scripts if available, these are old scripts and language setup was used with Jquery.i18 library.
<script src="\Jquery/i18n/jquery.i18n.min.js" type="text/javascript"></script>
<script src="\Jquery/i18n/jquery.i18n.emitter.bidi.js" type="text/javascript"></script>
<script src="\Jquery/i18n/i18n.custom.js" type="text/javascript"></script>

2. add i18n.custom.js file at this path - \Jquery/i18n/i18n.custom.js

3. add a script tag just above </body> ending tag. This will ensure that this script will load only after all the DOM elements are loaded on page load.
<script src="\Jquery/i18n/i18n.custom.js" type="text/javascript" defer></script>

4. add download button after starting <body> tag to download data-i18n attributes which are applied on elements in a json format.
<button id="downloadTranslations" class="d-none-">Download JSON</button>

5. add selectmenu to switch languages.
- Also change language option from here also as required.
<select class="dvLang">
    <option value="en">english</option>
    <option value="np">nepal</option>
    <option value="ar">arabic</option>
</select>

6. Add en.json, np.json and ar.json in Webclient/i18n/_ _ _.json as per requirement of languages.

7. Add required language css files in Webclient/Css/_ _ _.css folder.
- No need to add english.css because global.css and root.css are handling it.
- example: arabic.css or nepal.css or french.css

8. Add path as per requirement.
- languageStyles: {
    ar: { cssFile: "Css/arabic.css", direction: "rtl" },
    en: { cssFile: "Css/global.css", direction: "ltr" }
}
NOTE: there should be same css files and json files, which you want to use in the project. Please remove unnecessary files from Webclient/i18n/....json and from Webclient/Css/_ _ _.css folders related to language script.

9. Don't change anything inside the language script file except languageStyles:{} object.

10. Remove download button from SiteMaster.master if everything is done.

function BindBanner() {
    //var languageSelector = $("#languageSelector").val();
    try {
        var bannerHTML = "";
        var path = "";
        switch ($("#languageSelector").val()) {
            case "en":
                bannerHTML = '<%=HttpContext.Current.Application[string.Format("{0}banner",Convert.ToString(Session["CategoryName"]))]%>';
                break;
            case "np":
                bannerHTML = '<%=HttpContext.Current.Application[string.Format("{0}bannernp",Convert.ToString(Session["CategoryName"]))]%>';
                break;
            case "ar":
                bannerHTML = '<%=HttpContext.Current.Application[string.Format("{0}bannerar",Convert.ToString(Session["CategoryName"]))]%>';
                break;
        }
        if (bannerHTML != '') {
            appendBanner(bannerHTML);
        }
        else {
            switch ($("#languageSelector").val()) {
                case "en":
                    path = '<%=string.Format("{0}banner",Convert.ToString(Session["CategoryName"]))%>';
                    break;
                case "np":
                    path = '<%=string.Format("{0}bannernp",Convert.ToString(Session["CategoryName"]))%>';
                    break;
                case "ar":
                    path = '<%=string.Format("{0}bannerar",Convert.ToString(Session["CategoryName"]))%>';
                    break;
            }
            $.ajax({
                type: 'POST',
                url: 'Index.aspx/GetHomeBanner',
                contentType: 'application/json; charset=utf-8',
                dataType: 'json',
                data: "{path:'" + path + "'}",
                success: function (msg) {
                    if (msg.d != '') {
                        $("#dvHeroSlider").append(msg.d);
                        appendBanner(msg.d); // Assuming 'en' is the default
                    }
                }
            });
        }
    }
    catch (e) {
        console.error(e);
    }
}



// Function to append the banner based on the language
function appendBanner(data) {
    $("#dvHeroSlider").empty();
    $("#dvHeroSlider").append(data);



    // Initialize the Swiper
    const heroSlider = new Swiper(".dvHeroSlider.swiper", {
        direction: "horizontal",
        loop: false,
        speed: 500,
        slidesPerView: 1,
        navigation: {
            nextEl: ".swiper-button-next",
            prevEl: ".swiper-button-prev",
        },
        pagination: {
            el: ".swiper-pagination",
            clickable: true,
        },
        spaceBetween: 0,
        breakpoints: {
            0: {
                slidesPerView: 1,
            },
        },
        mousewheel: {
            enabled: false,
        },
        keyboard: {
            enabled: true,
        },
    });
}
